<h3 style="text-align:center;">K4 — graphe simple non orienté complet à 4 sommets</h3>
<div id="k4box" class="jxgbox" style="width:500px; height:500px; margin:0 auto;"></div>
<script>
  if (typeof JXG === 'undefined') {
    document.getElementById('k4box').innerHTML =
      '<p style="color:red;text-align:center;">JSXGraph n\'a pas pu se charger.</p>';
  } else {
    var brd = JXG.JSXGraph.initBoard('k4box', {
      boundingbox: [-5, 5, 5, -5],
      axis: false,
      showCopyright: false,
      showNavigation: false
    });

    // 4 sommets déplaçables, disposés pour une lecture planaire (un sommet au centre)
    var A = brd.create('point', [0, 3.5], {name: 'A', size: 4, fillColor: '#026573', strokeColor: '#026573'});
    var B = brd.create('point', [-3, -2], {name: 'B', size: 4, fillColor: '#026573', strokeColor: '#026573'});
    var C = brd.create('point', [3, -2], {name: 'C', size: 4, fillColor: '#026573', strokeColor: '#026573'});
    var D = brd.create('point', [0, 0.2], {name: 'D', size: 4, fillColor: '#e07b00', strokeColor: '#e07b00'});

    var pts = [A, B, C, D];
    var edgeStyle = {strokeColor: '#333333', strokeWidth: 2, fixed: true, straightFirst: false, straightLast: false};

    // Les 6 arêtes de K4 : toutes les paires de sommets sont reliées
    for (var i = 0; i < pts.length; i++) {
      for (var j = i + 1; j < pts.length; j++) {
        brd.create('segment', [pts[i], pts[j]], edgeStyle);
      }
    }

    brd.create('text', [-4.7, 4.6], {
      text: '4 sommets, 6 arêtes — déplacez les points',
      fontSize: 13, strokeColor: '#555555', fixed: true
    });
  }
</script>
