<h1>Riemann Sums</h1>

    <div class="description">
        Approximate the integral of the sine function over a configurable interval controlled by sliders
        for the start and end points. Use the "type" slider to pick the Riemann sum method. A Riemann sum
        with a selectable number of subdivisions is drawn under the curve, and its computed value is
        displayed dynamically alongside the live exact value of the integral.
    </div>

    <script type="text/javascript">
        (function () {
            if (typeof JXG === 'undefined') {
                document.getElementById('riemann-error').style.display = 'block';
                return;
            }

            var board = JXG.JSXGraph.initBoard('jxgbox', {
                axis: true,
                boundingbox: [-8, 4, 8, -4]
            });

            var sumTypes = ['lower', 'upper', 'left', 'right', 'middle', 'trapezoidal', 'simpson'];

            // Slider for number of subdivisions (n)
            var s = board.create('slider', [[1, 3.5], [5, 3.5], [1, 10, 50]], {
                name: 'n', snapWidth: 1
            });

            // Slider for interval start
            var a = board.create('slider', [[1, 3], [5, 3], [-10, -2 * Math.PI, 0]], {
                name: 'start'
            });

            // Slider for interval end
            var b = board.create('slider', [[1, 2.5], [5, 2.5], [0, Math.PI, 10]], {
                name: 'end'
            });

            // Slider to pick the Riemann sum type (snapped to an index into sumTypes).
            // Using a native JSXGraph slider instead of an HTML <select> keeps this
            // entirely inside the board's own script -- no separate DOM control with
            // an inline onChange handler that could get stripped independently.
            var t = board.create('slider', [[1, 2], [5, 2], [0, 2, sumTypes.length - 1]], {
                name: 'type', snapWidth: 1
            });

            board.create('text', [1, 2.35, function () {
                return 'Sum type: ' + sumTypes[Math.round(t.Value())];
            }], { fontSize: 13, color: '#495057' });

            var f = function (x) { return Math.sin(x); };

            var plot = board.create('functiongraph', [f, function () { return a.Value(); }, function () { return b.Value(); }]);

            var os = board.create('riemannsum', [
                f,
                function () { return s.Value(); },
                function () { return sumTypes[Math.round(t.Value())]; },
                function () { return a.Value(); },
                function () { return b.Value(); }
            ], {
                fillColor: '#ffff00',
                fillOpacity: 0.3
            });

            board.create('text', [-7.8, -3.2, function () {
                return 'Riemann sum ≈ ' + (JXG.Math.Numerics.riemannsum(
                    f, s.Value(), sumTypes[Math.round(t.Value())], a.Value(), b.Value()
                )).toFixed(4);
            }], { fontSize: 14, color: '#495057' });

            board.create('text', [-7.8, -3.6, function () {
                var exact = Math.cos(a.Value()) - Math.cos(b.Value());
                return 'Exact value = ∫ sin(x) dx = ' + exact.toFixed(4);
            }], { fontSize: 12, color: '#868E96' });
        })();
    </script>
