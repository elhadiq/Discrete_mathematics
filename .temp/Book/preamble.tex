% ---------- exercises & solutions ----------
\newcounter{exercise}[chapter]
\renewcommand{\theexercise}{\thechapter.\arabic{exercise}}
\newenvironment{exercise}[1]{% #1 = \easy | \medium | \hard
  \refstepcounter{exercise}\par\medskip\needspace{4\baselineskip}%
  \noindent{\bfseries\color{DarkBlue}Exercise \theexercise}\ #1\par\nopagebreak\smallskip
  \noindent\ignorespaces}{\par\medskip}
\newenvironment{solution}[1]{% #1 = exercise number, e.g. 2.3
  \par\medskip\needspace{3\baselineskip}%
  \noindent{\bfseries\color{Orange}Solution #1.}\ \ignorespaces}{%
  \hfill$\blacksquare$\par\medskip}
